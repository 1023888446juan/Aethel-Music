create table usuario(
id_usuario int auto_increment primary key,
nombre_usuario varchar(50),
correo_usuario varchar (50),
contraseña_usuario varchar(50),
rol_usuario varchar (20) default 'usuario' 'admin' 'dj'
);

create table playlist ( 
id_playlist int auto_increment primary key not null,
nombre_playlist varchar(30),
descripcion_playlist varchar(80),
fecha_creacion_playlist date,
id_usuario int
);

create table playlist_pista(
id_playlist_pista int auto_increment primary key not null,
id_playlist int,
id_pista int
);

create table pista(
id_pista int auto_increment primary key not null,
titulo_pista varchar (30),
duracio_pista time,
formato_pista varchar (30)
);

create table descarga (
id_descarga int auto_increment primary key not null,
fecha_descarga date,
genero_descarga varchar (30),
id_usuario_fk int
);

create table DJ_seccion(
id_secciondj int auto_increment primary key,
descripcion_dj_seccion varchar(50),
id_usuario int
);

create table Efecto_DJSeccion(
id_secciondj int,
id_efecto int
);

create table efectos(
id_efecto int auto_increment primary key not null,
titulo varchar(30),
duracion time,
formato varchar(30)
);

create table Configuracion(
id_configuracion int auto_increment primary key not null,
tamaño_configuracion varchar(30),
color_configuracion varchar (30),
idioma_configuracion varchar (30),
notificacion_configuracion varchar (30),
id_usuario int
);



create table favorito (
id_favorito  int auto_increment primary key not null,
fecha_agregado_favorito date,
nombre_favorito varchar(30),
id_usuario int
);

create table Historial_reproducion(
id_historial int auto_increment primary key not null,
tiempo_escuchado_historial_reproducion  bigint,
fecha_historial_reproducion date,
id_usuario int
);

create table Album (
id_album int auto_increment primary key not null,
titulo_album varchar(30),
descripcion_album varchar(30),
id_pista int
);
 
create table Artistas(
id_artista int auto_increment primary key not null,
nombre_artista varchar(30),
id_album int
); 

create table Genero(
id_genero int auto_increment primary key not null,
nombre_genero varchar(30), 
descripcion_genero varchar(50),
id_pista int
);

alter table Efecto_DJSeccion add foreign key (id_efecto) references efectos(id_efecto);

SELECT 
    u.id_usuario,
    u.nombre_usuario,
    u.rol_usuario,
    pl.id_playlist,
    pl.nombre_playlist,
    pl.fecha_creacion_playlist,
    pp.id_pista,
    p.id_pista,
    p.titulo_pista,
    p.duracio_pista,
    p.formato_pista
FROM    
        usuario AS u, 
        /*descarga AS d,*/
        playlist AS pl, 
        playlist_pista AS pp,
        pista AS p 
where /*u.id_usuario = d.id_usuario_fk 
and*/   u.id_usuario = pl.id_usuario 
and   pl.id_playlist = pp.id_playlist 
and   p.id_pista = pp.id_pista /*
and   pl.id_usuario = d.id_usuario_fk
and   u.rol_usuario = 'dj'; */
